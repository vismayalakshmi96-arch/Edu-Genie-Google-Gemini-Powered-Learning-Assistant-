# -------------------------------------------------
# GEMINI AI FUNCTION
# -------------------------------------------------

def ask_gemini(question, mode):

    if not client:
        return "Gemini API key is not configured. Please set GEMINI_API_KEY."

    prompt = f"""
You are EduGenie, an AI-powered educational assistant.

Student Question:
{question}

Learning Mode:
{mode}

Instructions:
- Give accurate and educational information.
- Use simple language suitable for college students.
- Explain difficult concepts clearly.
- Use headings and bullet points when useful.
- If the mode is Exam Answer, provide a well-structured answer.
- If the mode is Short Notes, keep the answer concise.
- If the mode is Step-by-Step, explain the solution step by step.
- Do not unnecessarily complicate the answer.
"""

    try:

        response = client.models.generate_content(
            model="gemini-2.5-flash",
            contents=prompt
        )

        return response.text

    except Exception as e:

        return "Error while contacting Gemini: " + str(e)


# -------------------------------------------------
# MAIN ROUTE
# -------------------------------------------------

@app.route("/", methods=["GET", "POST"])
def home():

    answer = ""
    question = ""

    if request.method == "POST":

        question = request.form.get("question", "")
        mode = request.form.get(
            "mode",
            "Simple Explanation"
        )

        if question.strip():

            answer = ask_gemini(
                question,
                mode
            )

    return render_template_string(
        HTML,
        answer=answer,
        question=question
    )
